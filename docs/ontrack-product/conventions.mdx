import { CardGrid, LinkCard } from '@astrojs/starlight/components';

export const track = trackNamed('ontrack-product');

The conventions of ontrack-product {track.version}, the release ontrack depends
on, as a repository receives them: `proposed`, their traits empty, the
recommended values in each body. They are also in
[the track's repository](https://github.com/trivoallan/ontrack-product/tree/main/adr).

{track.domains.map((domain) => (

    <div class="convention-cards">
      <CardGrid>
        {domain.conventions.map((q) => (
          <LinkCard title={`${q.adr} ${q.title}`} href={`../${q.id}/`} />
        ))}
      </CardGrid>
    </div>
  </>
))}
